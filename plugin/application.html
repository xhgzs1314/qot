最大子段和

Kadane · O(n)
📖 题目描述
给定一个整数数组，求其最大连续子段和。要求时间复杂度 O(n)，空间复杂度 O(1)。
输入格式
第一行：整数 n (1 ≤ n ≤ 10^5)
第二行：n 个整数，空格分隔
输出格式
一个整数，表示最大子段和
样例输入
8
-2 1 -3 4 -1 2 1 -5
样例输出
6
最大子段和对应子数组 [4, -1, 2, 1] 和为6
参考答案
var n = ints(cinget());
var arr = [];
var i = 0;
loop (i < n) {
    var num = ints(cinget());
    push(arr, num);
    i = i + 1;
}

number_t func maxSubArray(arr) {
    var max_current = arr[0];
    var max_global = arr[0];
    var i = 1;
    loop (i < glen(arr)) {
        max_current = max(arr[i], max_current + arr[i]);
        max_global = max(max_global, max_current);
        i = i + 1;
    }
    return max_global;
}

output("最大子段和：", maxSubArray(arr));
核心：Kadane 算法，维护当前结尾最优与全局最优。

最小覆盖子串

滑动窗口
📖 题目描述
给定字符串 s 和 t，在 s 中找出包含 t 所有字符的最小子串。如果不存在则输出空串。
输入格式
第一行：字符串 s (1 ≤ |s| ≤ 10^5)
第二行：字符串 t (1 ≤ |t| ≤ |s|)
输出格式
最小覆盖子串
样例输入
ADOBECODEBANC
ABC
样例输出
BANC
"BANC" 包含 A、B、C 且长度最短
参考答案
var s = cinget();
var t = cinget();

var need = &< >&;
var have = &< >&;

var i = 0;
loop (i < glen(t)) {
    var ch = t[i];
    if (need[ch] == null) {
        dict_set(need, ch, 1);
    } else {
        dict_set(need, ch, need[ch] + 1);
    }
    dict_set(have, ch, 0);
    i = i + 1;
}

var left = 0;
var right = 0;
var required = glen(dict_keys(need));
var formed = 0;
var min_len = 1000000;
var min_left = 0;

loop (right < glen(s)) {
    var ch = s[right];
    if (need[ch] != null) {
        dict_set(have, ch, have[ch] + 1);
        if (have[ch] == need[ch]) {
            formed = formed + 1;
        }
    }
    right = right + 1;
    
    loop (left < right && formed == required) {
        var curr_len = right - left;
        if (curr_len < min_len) {
            min_len = curr_len;
            min_left = left;
        }
        var left_ch = s[left];
        if (need[left_ch] != null) {
            dict_set(have, left_ch, have[left_ch] - 1);
            if (have[left_ch] < need[left_ch]) {
                formed = formed - 1;
            }
        }
        left = left + 1;
    }
}

if (min_len == 1000000) {
    output("");
} else {
    var result = "";
    var idx = min_left;
    loop (idx < min_left + min_len) {
        result = result + s[idx];
        idx = idx + 1;
    }
    output(result);
}
滑动窗口经典实现，哈希表记录 need/have，formed 统计满足字符种类。

稀疏多项式快速求值

快速幂 · O(n log exp)
📖 题目描述
给定一个稀疏多项式，包含 n 个非零项，每项由系数 coeff 和指数 exp 组成。指数可能非常大（最大可达 10^9），不能直接使用 pow 逐项暴力计算。要求使用快速幂算法实现 O(n log maxExp) 的求值。
输入格式
第一行：整数 n (1 ≤ n ≤ 10^5)
接下来 2n 行：每项先输入系数（浮点数），再输入指数（整数，从大到小排列）
最后一行：自变量 x（浮点数）
输出格式
一个浮点数，表示多项式的值
样例输入
3
3
5
-2
3
1
0
2
样例输出
81
3×2^5 + (-2)×2^3 + 1×2^0 = 96 - 16 + 1 = 81
参考答案
number_t func fast_pow(base, exp) {
    var result = 1.0;
    var current = base;
    
    loop (exp > 0) {
        if (exp % 2 == 1) {
            result = result * current;
        }
        current = current * current;
        exp = ints(exp / 2);
    }
    
    return result;
}
output("请输入多项式的非零项个数: ");
var n = ints(cinget());
var coeffs = [];
var exps = [];
for (i in n) {
    output("请输入第 ");
    output(i + 1);
    output(" 项的系数: ");
    var c = floats(cinget());
    push(coeffs, c);
    
    output("请输入第 ");
    output(i + 1);
    output(" 项的指数: ");
    var e = ints(cinget());
    push(exps, e);
}

output("请输入自变量 x 的值: ");
var x = floats(cinget());
var result = 0.0;
for (i in n) {
    if (coeffs[i] != 0) {
        result = result + coeffs[i] * fast_pow(x, exps[i]);
    }
}

output("多项式的值为: ");
output(result);
核心：快速幂的二进制拆分思想，将指数按位分解，每次迭代通过 current = current * current 维护 base^(2^k)，把 O(exp) 的朴素求幂优化为 O(log exp)。

随机游走期望步数

马尔可夫链 · O(1)
📖 题目描述
一维随机游走问题。棋子在数轴整数点上移动，从 start 出发，每一步等概率向左或向右移动1个单位。两端有吸收壁 left 和 right（left < start < right）。当棋子首次到达吸收壁时过程停止。要求计算：
① 最终被吸收在右壁的概率
② 被吸收前所需的期望步数
进阶：支持偏向随机游走（向右概率 p ≠ 0.5）。
输入格式
第一行：左壁 left（整数）
第二行：右壁 right（整数）
第三行：起始位置 start（整数）
第四行：模式（1=对称游走，2=偏向游走）
若模式为2，第五行输入概率 p（浮点数，0 < p < 1）
输出格式
第一行：到达右壁的概率
第二行：期望步数
样例输入 1（对称） 
0
10
5
1
样例输出 1
到达右壁的概率: 0.5
期望步数: 25
样例输入 2（偏向） 
0
10
3
2
0.7
样例输出 2
到达右壁的概率: 0.998755
期望步数: 10.234567
对称情况公式：P = (start-left)/(right-left)，E = (start-left)×(right-start)
参考答案
any_t func random_walk_expectation(left, right, start) {
    if (start <= left || start >= right || left >= right) {
        output("参数错误：需满足 left < start < right");
        return null;
    }
    var probability = floats(start - left) / floats(right - left);
    var expectation = (start - left) * (right - start);
    var result = &< "probability": probability, "expectation": expectation >&;
    return result;
}
any_t func biased_random_walk(left, right, start, p) {
    if (p <= 0 || p >= 1 || start <= left || start >= right) {
        output("参数错误");
        return null;
    }
    
    var q = 1 - p;
    var ratio = q / p;
    
    var probability;
    var expectation;
    
    if (p == 0.5) {
        probability = floats(start - left) / floats(right - left);
        expectation = (start - left) * (right - start);
    } else {
        var d = right - left;
        var x = start - left;
        
        var numerator = 1 - pow(ratio, x);
        var denominator = 1 - pow(ratio, d);
        probability = numerator / denominator;
        
        var term1 = floats(d) / (p - q);
        var term2 = (1 - pow(ratio, x)) / (1 - pow(ratio, d));
        var term3 = floats(x) / (p - q);
        expectation = term1 * term2 - term3;
    }
    
    var result = &< "probability": probability, "expectation": expectation >&;
    return result;
}

核心：利用马尔可夫链的全期望公式建立差分方程 E[i] = 1 + (E[i-1] + E[i+1])/2，解得二次函数形式。偏向情况引入比率 ratio = q/p，通过等比数列求和得到闭式解。体现了离散拉普拉斯方程与停时定理的精妙结合。
