Qot-A002

var students = [["张三", 90], ["李四", 85], ["王五", 78]];
any_t func add_score(arr, name, score) {
    push(arr, [name, score]);
    return true;
}
number_t func average_score(arr) {
    var sums = 0;
    var count = glen(arr);
    for (i in count-1) {
        var student = arr[i];  
        sums = sums + student[1];  
    }
    return sums / count;
}


for (i in glen(students)-1) {
    var student = students[i];  
    output("姓名：", student[0], "，成绩：", student[1]);  
}

add_score(students, "赵六", 92);

var avg = average_score(students);
output("平均分：", avg);
if (avg > 85) {
    output("优秀");
} else {
    output("待提升");
}

Qot-A004 音乐播放

require "miaudio";
var player = miaudio##MusicPlayer_Create();
var result = miaudio##MusicPlayer_Open(player, "chengnan.mp3"); //城南花已开...
if (result == 0) {
    output("打开文件失败！");
    return false;
}
miaudio##MusicPlayer_Play(player);
loop (true) {
    var state = miaudio##MusicPlayer_GetState(player);
    var position = miaudio##MusicPlayer_GetPosition(player);
    var duration = miaudio##MusicPlayer_GetDuration(player);
    if (state == 3 || position >= duration) {
        output("\n播放结束！");
        stop;
    }
    sleep(100);
}
miaudio##MusicPlayer_Destroy(player);

Qot-A006 天气API调用

//Thanks to uapis.cn
http_set_user_agent("QOT-WeatherClient/1.0");
http_set_verify_ssl(0);
output("请输入城市名称: ");
var city = cinget();
var url = "https://uapis.cn/api/v1/misc/weather?city=" + city;
output("正在查询 " + city + " 的天气...");
var result = http_send(url, null, "GET", 10000);
var statusCode = result[0];
var responseData = result[2];
if (statusCode == 200) {
    var weatherData = json_decode(responseData);
    if (weatherData != null) {
        output("========== 天气查询结果 ==========");
        output("城市: " + weatherData["city"]);
        output("天气状况: " + weatherData["weather"]);
        output("温度: " + strings(weatherData["temperature"]) + "°C");
        output("风向: " + weatherData["wind_direction"]);
        output("风力: " + weatherData["wind_power"]);
        output("湿度: " + strings(weatherData["humidity"]) + "%");
        output("发布时间: " + weatherData["report_time"]);
        output("=================================");
        // 给出建议
        var condition = weatherData["weather"];
        if (strrep(condition, "雨", "") != condition) {
            output("💡 建议：有雨，出门请带伞");
        } eif (strrep(condition, "雪", "") != condition) {
            output("💡 建议：下雪，注意保暖和出行安全");
        } eif (strrep(condition, "晴", "") != condition) {
            output("💡 建议：天气晴朗，适合户外活动");
        } eif (strrep(condition, "云", "") != condition) {
            output("💡 建议：多云天气，适宜出行");
        } else {
            output("💡 建议：请关注天气变化");
        }
    } else {
        output("解析天气数据失败");
    }
} else {
    output("查询失败，HTTP状态码: " + strings(statusCode));
    if (responseData != null && responseData != "") {
        output("错误信息: " + responseData);
    }
}
