    数据库操作Qot-A001

    用 FFI 加载 libmysql.dll，按 mysql_real_connect / mysql_query 的签名连接并执行 SQL。

    // FFI 版：需将 libmysql.dll 置于同目录
var mysql = ffi_load("libmysql.dll");
var linkings = "555555:555555@tcp(localhost:3306)/555555?charset=utf8mb4";
var conn = ffi_call_v(ffi_sym(mysql, "mysql_init"), ["ptr"], [0]);
// 原 MySQLConnect(linkings)：连接串按 libmysql 参数拆分（user:pass@tcp(host:port)/db）
ffi_call_v(ffi_sym(mysql, "mysql_real_connect"),
           ["ptr", "ptr", "ptr", "ptr", "ptr", "i32", "ptr", "i32"],
           [conn, ffi_ptr("localhost"), ffi_ptr("555555"), ffi_ptr("555555"),
            ffi_ptr("555555"), 3306, 0, 0]);
// 原 MySQLExecute(conn, "DELETE ...")
ffi_call_v(ffi_sym(mysql, "mysql_query"), ["i32", "ptr", "ptr"],
           [conn, ffi_ptr("DELETE FROM users WHERE username = 'admin';")]);

    数组与函数Qot-A002

    二维数组 + 自定义函数：遍历输出、追加元素、求平均分并按条件分支。

    var students = [["张三", 90], ["李四", 85], ["王五", 78]];
any_t func add_score(arr, name, score) {
    push(arr, [name, score]);
    return true;
}
number_t func average_score(arr) {
    var sums = 0;
    var count = glen(arr);
    for (i in count-1) {
        var student = arr[i];  
        sums = sums + student[1];  
    }
    return sums / count;
}


for (i in glen(students)-1) {
    var student = students[i];  
    output("姓名：", student[0], "，成绩：", student[1]);  
}

add_score(students, "赵六", 92);

var avg = average_score(students);
output("平均分：", avg);
if (avg > 85) {
    output("优秀");
} else {
    output("待提升");
}

    音乐播放Qot-A004

    miaudio.dll 播放 MP3：类型码必须用 i32，且要等状态机真正进入播放中再跟踪结束。

    // miaudio 播放示例
// —— 类型码必须用 "i32"；用 "f64" 会按 XMM0 读 int 返回值（x64 上 int 在
// EAX），拿到的是垃圾（实测恒 0）→ `position >= duration` 变 `0 >= 0`
// 立即「播放结束」。返回指针的 Create 用 "ptr"（无类型码 ffi_call 按 f64
// 读会把指针位模式当小数，后续当 ptr 传回被拒）
// 【状态机】0=未知 1=播放中 2=已暂停 3=已停止（Open 后未 Play 时也是 3）
// —— 循环必须先等 state 进入 1/2 再等 3
var miaudio = ffi_load("package/miaudio/miaudio.dll");
any_t func MusicPlayer_Create() {
    return ffi_call_v(ffi_sym(miaudio, "MusicPlayer_Create"), ["ptr"], []);
}
any_t func MusicPlayer_Open(p, f) {
    return ffi_call_v(ffi_sym(miaudio, "MusicPlayer_Open"), ["i32", "ptr", "ptr"], [p, ffi_ptr(f)]);
}
any_t func MusicPlayer_Play(p) {
    ffi_call_v(ffi_sym(miaudio, "MusicPlayer_Play"), ["void", "ptr"], [p]);
}
any_t func MusicPlayer_GetState(p) {
    return ffi_call_v(ffi_sym(miaudio, "MusicPlayer_GetState"), ["i32", "ptr"], [p]);
}
number_t func MusicPlayer_GetPosition(p) {
    return ffi_call_v(ffi_sym(miaudio, "MusicPlayer_GetPosition"), ["i32", "ptr"], [p]);
}
number_t func MusicPlayer_GetDuration(p) {
    return ffi_call_v(ffi_sym(miaudio, "MusicPlayer_GetDuration"), ["i32", "ptr"], [p]);
}
any_t func MusicPlayer_Destroy(p) {
    ffi_call_v(ffi_sym(miaudio, "MusicPlayer_Destroy"), ["void", "ptr"], [p]);
}

var player = MusicPlayer_Create();
if (player == null) {
    output("创建播放器失败！");
    output(ffi_last_error());
    return false;
}
var result = MusicPlayer_Open(player, "ot-asset/audio/chengnan.mp3"); //城南花已开...
if (result == 0 || result == null) {
    output("打开文件失败！");
    output(ffi_last_error());
    MusicPlayer_Destroy(player);
    return false;
}
MusicPlayer_Play(player);

// 启动确认：等 state 进入「播放中(1)/暂停(2)」，2 秒窗口
var started = false;
var guard = 0;
loop (started == false && guard < 20) {
    var st = MusicPlayer_GetState(player);
    if (st == 1 || st == 2) {
        started = true;
    }
    guard = guard + 1;
    time_sleep_ms(100);
}
if (started == false) {
    output("播放器启动失败（state 未进入播放中）！");
    MusicPlayer_Destroy(player);
    return false;
}

// 跟踪到停止（3）；position >= duration 兜底
loop (true) {
    var state = MusicPlayer_GetState(player);
    var position = MusicPlayer_GetPosition(player);
    var duration = MusicPlayer_GetDuration(player);
    if (position == null || duration == null) {
        output("\n获取播放状态失败！");
        output(ffi_last_error());
        stop;
    }
    if (state == 3 || position >= duration) {
        output("\n播放结束！");
        stop;
    }
    time_sleep_ms(500);
}
MusicPlayer_Destroy(player);

    天气 API 调用Qot-A006

    HTTP GET uapis.cn 天气接口，json_decode 后输出字段并给出出行建议（Thanks to uapis.cn）。

    //Thanks to uapis.cn
http_set_user_agent("QOT-WeatherClient/1.0");
http_set_verify_ssl(0);
output("请输入城市名称: ");
var city = cinget();
var url = "https://uapis.cn/api/v1/misc/weather?city=" + city;
output("正在查询 " + city + " 的天气...");
var result = http_send(url, null, "GET", 10000);
var statusCode = result[0];
var responseData = result[2];
if (statusCode == 200) {
    var weatherData = json_decode(responseData);
    if (weatherData != null) {
        output("========== 天气查询结果 ==========");
        output("城市: " + weatherData["city"]);
        output("天气状况: " + weatherData["weather"]);
        output("温度: " + strings(weatherData["temperature"]) + "°C");
        output("风向: " + weatherData["wind_direction"]);
        output("风力: " + weatherData["wind_power"]);
        output("湿度: " + strings(weatherData["humidity"]) + "%");
        output("发布时间: " + weatherData["report_time"]);
        output("=================================");
        // 给出建议
        var condition = weatherData["weather"];
        if (strrep(condition, "雨", "") != condition) {
            output("建议：有雨，出门请带伞");
        } eif (strrep(condition, "雪", "") != condition) {
            output("建议：下雪，注意保暖和出行安全");
        } eif (strrep(condition, "晴", "") != condition) {
            output("建议：天气晴朗，适合户外活动");
        } eif (strrep(condition, "云", "") != condition) {
            output("建议：多云天气，适宜出行");
        } else {
            output("建议：请关注天气变化");
        }
    } else {
        output("解析天气数据失败");
    }
} else {
    output("查询失败，HTTP状态码: " + strings(statusCode));
    if (responseData != null && responseData != "") {
        output("错误信息: " + responseData);
    }
}

    类 · 协议 · 宏Qot-A007

    class 的协议方法（__add__ / __eq__ / __to_str）让自定义类型参与运算与输出；外加编译期宏与官方 swap!。

    // 协议方法：让自定义类型参与运算与输出
class Vec2 {
    var x = 0;
    var y = 0;
    func init(ax, ay) { this->x = ax; this->y = ay; }
    func __add__(o) { return Vec2(this->x + o->x, this->y + o->y); }
    func __eq__(o) { return this->x == o->x && this->y == o->y ? 1 : 0; }
    func __to_str() { return "(" + int_to_str(this->x) + "," + int_to_str(this->y) + ")"; }
}

output(Vec2(1, 2) + Vec2(10, 20));    // (11,22)
output(Vec2(1, 2) == Vec2(1, 2));     // 1

// 编译期宏：把重复的代码形状交给编译器
macro twice(x) {
    return < "kind": "binary_op", "op": "+", "left": x, "right": x >&;
}
output(twice!(21));                   // 42

// 官方宏：真交换两个变量
var a = 1;
var b = 2;
swap!(a, b);
output(int_to_str(a));               // 2
output(int_to_str(b));               // 1

    标准库：CSV 统计Qot-A008

    require std/text.qot 与 std/data.qot：解析 CSV、分组求和、计数与文本清洗。

    require "std/text.qot";
require "std/data.qot";

var csv = "city,amount,kind
beijing,100,a
shanghai,250,b
beijing,50,c
";
var rows = csv_parse(csv);

// 按城市分组求和
var by_city = group_sum(rows, 0, 1);
output(by_city["beijing"]);          // 150
output(by_city["shanghai"]);         // 250

// 取第一列并做计数
output(group_count(rows, 2)["a"]);   // 1

// 清洗与折行
var cleaned = trim_all(rows);
output(normalize("  hello    world  "));   // hello world
output(wrap("alpha beta gamma", 11));

    文件处理流水线Qot-A009

    逐行读取 → 清洗计数 → 写回 clean.txt，标准库 fs + text 的组合用法。

    require "std/fs.qot";
require "std/text.qot";

var lines = read_lines("data.txt");
if (lines == null) {
    output("读取失败");
} else {
    var total = 0;
    for (ln in lines) {
        var t = trim_all(ln);
        if (t != "") {
            total = total + 1;
        }
    }
    output("有效行数: " + int_to_str(total));
    write_lines("clean.txt", lines);
}

    GUI：打招呼Qot-A010

    vg_* 原语搭窗口 + 事件循环：点击按钮读取输入框内容并回写标签。

    完整源码见 testing/gui/hello_gui.qot（可用 python index.py run 运行，也可 python index.py aot 编译成单文件 exe——两引擎行为逐位一致）。核心骨架：

    var w = vg_create_window("你好 · GUI", 520, 340);
var name_box = vg_create_textbox(w, "在这里输入名字...", 40, 104, 300, 34);

var running = 1;
loop (running == 1) {
    var ev = vg_wait_event(15);              // mainloop 的心跳
    if (ev == null) { continue; }
    var id = ev[0];
    var kind = ev[1];                        // 1 点击 / 2 改变 / 3 选中 / 9 关闭
    if (kind == 9) { running = 0; }
    if (kind == 1 && id == btn_hello) {
        var name = vg_textbox_get_text(name_box);      // entry.get()
        if (name == null || name == "") {
            vg_label_set_text(result, "请先输入名字");
        } else {
            vg_label_set_text(result, "你好, " + name + "！");
        }
    }
}
